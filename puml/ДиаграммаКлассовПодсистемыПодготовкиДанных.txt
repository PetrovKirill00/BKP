class "Источник данных MOEX" as MoexSource {
  +get_securities_json()
  +read_secids()
  +download_candles_for_secid(secid, start_date, end_date)
  +download_secids(secids)
}

class "Фильтр ликвидных инструментов" as LiquidityFilter {
  +build_candles_summary(candles_dir)
  +filter_by_first_date(summary)
  +filter_by_active_months(summary)
  +filter_by_avg_rows(summary)
}

class "Исходные минутные данные" as RawData {
  -secid
  -begin
  -end
  -open
  -high
  -low
  -close
  -volume
  -value
}

class "Конвейер предварительной обработки" as PreprocessingPipeline {
  +read_liquid_secids()
  +read_secid_file(secid)
  +build_minute_grid(df)
  +restore_end(df)
}

class "Формирователь признаков" as FeatureBuilder {
  +add_basic_time_features(df)
  +add_cyclic_time_features(df)
  +add_log_features(df)
  +add_return_lag_features(df)
  +add_volume_features(df)
  +add_rolling_features(df)
  +add_volatility_features(df)
  +add_candle_shape_features(df)
}

class "Формирователь целевых значений" as TargetBuilder {
  +add_next_minute_targets(df)
  +add_target_return(df, horizon)
  +add_execution_returns(df, max_horizon)
  +add_hold_returns(df, max_horizon)
  +drop_rows_without_target(df)
}

class "Итоговый минутный датасет" as FinalDataset {
  -secid
  -begin
  -end
  -trade_date
  -features
  -targets
  +cast_dataset_dtypes(df)
  +build_dataset()
  +to_parquet(path)
}

MoexSource --> LiquidityFilter : CSV-файлы свечей
LiquidityFilter --> RawData : выбранные secid
RawData --> PreprocessingPipeline : исходные данные
PreprocessingPipeline --> FeatureBuilder : подготовленные данные
FeatureBuilder --> TargetBuilder : признаки
TargetBuilder --> FinalDataset : обучающий датасет

@enduml

top to bottom direction
skinparam shadowing false
skinparam classAttributeIconSize 0

class "Подсистема прогнозирования" as ForecastSubsystem {
  +fit_method(method, X, y)
  +make_prediction(method, X): ForecastResult
}

class "Окно признаков" as FeatureWindow {
  -features
  -timestamp: datetime
  -horizon: int
}

class "Результат прогноза" as ForecastResult {
  -expected_return: float
  -horizon: int
  -timestamp: datetime
}

class "Scaler" as Scaler {
  +fit(X)
  +transform(X)
  +fit_transform(X)
}

abstract class "Прогнозирующий метод" as ForecastMethod {
  +fit(X, y)
  +predict(X): ForecastResult
}

class "Методы,\nоснованные на правилах" as RuleBasedMethods {
  --
  Стратегия ценового импульса
  Стратегия возврата к среднему
  Стратегия тренда на основе
  скользящих средних
  Стратегия пробоя
  локального диапазона
}

class "Гребневая\nрегрессия" as RidgeRegression {
  --
  Ridge
}

class "Древовидные\nмодели" as TreeModels {
  --
  Решающее дерево
  Случайный лес
  CatBoost
}

class "Нейросетевые\nмодели" as NeuralModels {
  --
  TCN
  GRU
  LSTM
  Трансформер
}

ForecastSubsystem --> FeatureWindow : принимает
ForecastSubsystem --> ForecastMethod : использует
ForecastMethod --> ForecastResult : возвращает

ForecastMethod <|-- RuleBasedMethods
ForecastMethod <|-- RidgeRegression
ForecastMethod <|-- TreeModels
ForecastMethod <|-- NeuralModels

FeatureWindow --> ForecastMethod : передаётся на вход

Scaler --> RidgeRegression : масштабирует признаки
Scaler --> NeuralModels : масштабирует признаки
@enduml

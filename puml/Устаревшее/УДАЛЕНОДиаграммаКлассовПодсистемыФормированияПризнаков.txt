class "Подготовленные минутные данные" as PreparedMinuteDataset {
  -records
  +get_row_count()
}

class "Конвейер формирования признаков" as FeatureEngineeringPipeline {
  +add_basic_time_features(data)
  +add_cyclic_time_features(data)
  +add_return_lag_features(data)
  +add_volume_features(data)
  +add_rolling_features(data)
  +add_volatility_features(data)
  +add_candle_shape_features(data)
  +build_feature_vector(data): FeatureVector
}

class "Вектор признаков" as FeatureVector {
  -basic_time_features
  -cyclic_time_features
  -return_lag_features
  -volume_features
  -rolling_features
  -volatility_features
  -candle_shape_features
  +to_matrix()
}

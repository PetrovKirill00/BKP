@startuml
title UML-диаграма деятельности модуля принятия торговых решений

skinparam shadowing false
skinparam dpi 150
skinparam activity {
  FontSize 13
  BorderColor Black
  BackgroundColor White
}
skinparam defaultFontName Arial

start

:Получить данные текущей минуты:
прогнозы, цены, состояние портфеля;

:Увеличить время удержания
открытых позиций;

if (Есть открытые позиции?) then (да)
  :Проверить каждую позицию;

  if (Время удержания >= лимита
  или прогноз удержания недостаточен?) then (да)
    :Закрыть позицию
    по цене закрытия текущей минуты;
    :Зафиксировать продажу;
  else (нет)
    :Удерживать позицию;
  endif
endif

:Рассчитать свободные слоты
для новых позиций;

if (Есть свободные слоты
и доступный капитал?) then (да)
  :Отобрать кандидатов:
  прогноз входа >= порога входа
  и ожидаемая доходность > 0;

  if (Есть кандидаты?) then (да)
    :Выбрать лучших кандидатов
    по прогнозу входа;

    :Распределить капитал
    пропорционально
    ожидаемой доходности
    с учётом ограничений;

    :Открыть длинные позиции
    по цене открытия следующей минуты;

    :Зафиксировать покупку;
  else (нет)
    :Новые позиции не открывать;
  endif
else (нет)
  :Новые позиции не открывать;
endif

:Переоценить портфель;
:Сохранить кривую капитала
и журнал сделок;

stop
@enduml

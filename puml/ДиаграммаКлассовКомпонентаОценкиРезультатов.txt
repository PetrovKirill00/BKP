@startuml
title UML-диаграмма классов компонента оценки результатов

left to right direction
skinparam shadowing false
skinparam classAttributeIconSize 0
skinparam linetype ortho
skinparam nodesep 32
skinparam ranksep 28

class "Компонент оценки результатов" as EvaluationComponent {
  +calculate_backtest_metrics(...)
  +read_summary()
  +build_report()
}

class "Результаты исторического тестирования" as BacktestResults {
  -equity_history
  -trade_events
  -closed_trades
  -summary
}

class "Метрики портфеля" as PortfolioMetrics {
  -final_equity
  -total_return_pct
  -max_drawdown_pct
  -return_to_drawdown
  -time_in_market_share
}

class "Метрики сделок" as TradeMetrics {
  -trades
  -win_rate
  -profit_factor
  -avg_win_bp
  -avg_loss_bp
  -payoff_ratio
}

class "Метрики активности и издержек" as ActivityCostMetrics {
  -total_cost_pct
  -turnover
  -mean_holding_minutes
  -avg_open_positions
  -max_open_positions
}

class "Построитель графиков" as PlotBuilder {
  +plot_grouped_bars(...)
  +plot_equity_curves(...)
  +save_slide_1(...)
  +save_slide_2(...)
}

class "Итоговый отчёт" as ResultReport {
  -tables
  -charts
  -conclusions
}

EvaluationComponent --> BacktestResults : использует
EvaluationComponent --> PortfolioMetrics : рассчитывает
EvaluationComponent --> TradeMetrics : рассчитывает
EvaluationComponent --> ActivityCostMetrics : рассчитывает

PortfolioMetrics --> ResultReport : включаются
TradeMetrics --> ResultReport : включаются
ActivityCostMetrics --> ResultReport : включаются

BacktestResults --> PlotBuilder : передаются
PlotBuilder --> ResultReport : добавляет графики

note bottom of PlotBuilder
  Графики строятся для разных уровней
  торговых издержек: 0,01%, 0,03% и 0,05%.
end note
@enduml

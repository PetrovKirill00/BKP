class "Загрузчик данных" as DataLoader {
  +load_bars(): List<MinuteBar>
  +load_liquid_stocks(): List<Instrument>
  +parse_response(raw_data): RawMinuteDataset
}

class "Инструмент" as Instrument {
  -secid: str
  -name: str
  -is_liquid: bool
}

class "Минутная свеча" as MinuteBar {
  -secid: str
  -begin: datetime
  -end: datetime
  -open: float
  -high: float
  -low: float
  -close: float
  -volume: float
  -value: float
  +is_valid(): bool
}

class "Набор сырых минутных данных" as RawMinuteDataset {
  -records
  -instruments
  +get_row_count()
}

class "Конвейер предварительной обработки" as PreprocessingPipeline {
  +preprocess(data): PreparedMinuteDataset
  +validate(data): bool
}

class "Проверка схемы и типов" as SchemaValidator {
  +check_required_columns(data): bool
  +check_column_types(data): bool
  +convert_types(data): RawMinuteDataset
}

class "Сортировка и удаление дубликатов" as SortAndDeduplicate {
  +check_sorting(data): bool
  +sort_by_begin(data): RawMinuteDataset
  +remove_duplicates(data): RawMinuteDataset
}

class "Разбиение по торговым дням" as TradeDaySplitter {
  +add_trade_date(data): RawMinuteDataset
  +split_by_trade_date(data)
}

class "Построитель минутной сетки" as MinuteGridBuilder {
  +build_minute_grid(data): PreparedMinuteDataset
  +count_missing_minutes(data): int
}

class "Заполнитель пропусков" as GapFiller {
  +fill_gaps_forward(data): PreparedMinuteDataset
  +mark_imputed_rows(data): PreparedMinuteDataset
}

class "Восстановление служебных полей" as MetadataRestorer {
  +restore_end(data): PreparedMinuteDataset
  +check_end_consistency(data): bool
}

class "Набор подготовленных минутных данных" as PreparedMinuteDataset {
  -records
  -trade_date
  -missing_rows_count
  +get_trade_dates()
}

class "Подготовленные минутные данные" as PreparedMinuteDataset {
  + records
}

class "Вектор признаков" as FeatureVector {
  + features
}

class "Конвейер формирования признаков" as FeatureEngineeringPipeline {
  + build_features(data): FeatureVector
}

class "Построитель базовых временных признаков" as BasicTimeFeatureBuilder {
  + add_basic_time_features(data): FeatureVector
}

class "Построитель циклических временных признаков" as CyclicTimeFeatureBuilder {
  + add_cyclic_time_features(data): FeatureVector
}

class "Построитель лагов доходности" as ReturnLagBuilder {
  + add_return_lags(data): FeatureVector
}

class "Построитель объёмных признаков" as VolumeFeatureBuilder {
  + add_log_volume_features(data): FeatureVector
  + add_log_value_features(data): FeatureVector
  + add_volume_diff_features(data): FeatureVector
}

class "Построитель скользящих характеристик" as RollingFeatureBuilder {
  + add_moving_average_features(data): FeatureVector
  + add_close_to_ma_features(data): FeatureVector
}

class "Построитель признаков изменчивости" as VolatilityFeatureBuilder {
  + add_volatility_features(data): FeatureVector
}

class "Построитель признаков геометрии свечи" as CandleShapeFeatureBuilder {
  + add_candle_shape_features(data): FeatureVector
}

PreparedMinuteDataset --> FeatureEngineeringPipeline : входные данные
FeatureEngineeringPipeline --> FeatureVector : формирует

FeatureEngineeringPipeline ..> BasicTimeFeatureBuilder : использует
FeatureEngineeringPipeline ..> CyclicTimeFeatureBuilder : использует
FeatureEngineeringPipeline ..> ReturnLagBuilder : использует
FeatureEngineeringPipeline ..> VolumeFeatureBuilder : использует
FeatureEngineeringPipeline ..> RollingFeatureBuilder : использует
FeatureEngineeringPipeline ..> VolatilityFeatureBuilder : использует
FeatureEngineeringPipeline ..> CandleShapeFeatureBuilder : использует

BasicTimeFeatureBuilder -[hidden]down-> CyclicTimeFeatureBuilder
CyclicTimeFeatureBuilder -[hidden]down-> ReturnLagBuilder
ReturnLagBuilder -[hidden]down-> VolumeFeatureBuilder
VolumeFeatureBuilder -[hidden]down-> RollingFeatureBuilder
RollingFeatureBuilder -[hidden]down-> VolatilityFeatureBuilder
VolatilityFeatureBuilder -[hidden]down-> CandleShapeFeatureBuilder

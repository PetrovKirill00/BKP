  -secid
  -begin
  -end
  -trade_date
  -is_imputed
  -open
  -high
  -low
  -close
  -volume
  -value
}

class "Конвейер формирования признаков" as FeaturePipeline {
  +add_basic_time_features(df)
  +add_cyclic_time_features(df)
  +add_log_features(df)
  +add_return_lag_features(df)
  +add_volume_features(df)
  +add_rolling_features(df)
  +add_volatility_features(df)
  +add_candle_shape_features(df)
}
